public method host:execute("subscribeTradeEvents", ...);

Brief

Subscribes a strategy or an indicator for trading events.

Declaration
Lua
string host:execute ("subscribeTradeEvents", cookie, table)

Parameters
cookie

(integer) The cookie to be sent to AsyncOperationFinished method of the indicator or strategy instance.
table

(string) the table to be subscribed for.

Can have the following values: "offers", "orders" or "trades".

See below for details when and how the AsyncOperationFinished method will be called for each kind of the messages.

"orders"

When an order appears or when order's status is changed.

The order identifier (orders.OrderID).

The new order status (orders.FixStatus).

The identifier of the request used to create the order.

"trades"

When a trade is opened or closed.

The trade identifier (trades.TradeID).

The order identifier of the order opened or closed the trade. (trades.OpenOrderID) for new trade or (closed trades.CloseOrderID) for closed trade.

The identifier of the request used to create the order. (trades.OpenOrderReqID) for new trade or (closed trades.CloseOrderReqID) for closed trade.

Note: The notification is sent asynchronously, so, depending on the strategy (indicator) performance and current load may be received later. So, the offer/order/trade object may be changed again before the notification is processed.

function Init()
    strategy:name("test trading events");
    strategy.parameters:addString("ACCOUNT", "Account", "", "");
    strategy.parameters:setFlag("ACCOUNT", core.FLAG_ACCOUNT);
    strategy.parameters:addInteger("TS", "Trail Stop", "", 1, 0, 200);
    strategy.parameters:addString("DIR", "Direction", "", "S");
end
 
local ACCOUNT;
local AMOUNT;
local OFFER;
 
 
function Prepare()
    ACCOUNT = instance.parameters.ACCOUNT;
    AMOUNT = core.host:execute("getTradingProperty", "baseUnitSize", instance.bid:instrument(), ACCOUNT);
    OFFER = core.host:findTable("offers"):find("Instrument", instance.bid:instrument()).OfferID;
    core.host:execute("subscribeTradeEvents", 1000, "offers");
    core.host:execute("subscribeTradeEvents", 1001, "orders");
    core.host:execute("subscribeTradeEvents", 1002, "trades");
end
 
local stop = false;
 
function Update()
    if not stop then
        local valuemap, success, msg;
 
        valuemap = core.valuemap();
        valuemap.OrderType = "OM";
        valuemap.OfferID = OFFER;
        valuemap.AcctID = ACCOUNT;
        valuemap.Quantity = AMOUNT;
        valuemap.BuySell = instance.parameters.DIR;
        valuemap.PegTypeStop = "O";
        if instance.parameters.DIR == "S" then
            valuemap.PegPriceOffsetPipsStop = 100;
        else
            valuemap.PegPriceOffsetPipsStop = -100;
        end
 
        valuemap.TrailStepStop = instance.parameters.TS;
        success, msg = terminal:execute(200, valuemap);
        assert(success, msg);
    end
end
 
function AsyncOperationFinished(cookie, success, message, message1, message2)
    if cookie == 200 then
        assert(success, message);
        stop = true;
    elseif cookie == 1000 then
        print("offer: %s", message);
    elseif cookie == 1001 then
        print("order: %s %s %s", message, message1, message2);
    elseif cookie == 1002 then
        print("trade: %s %s %s", message, message1, message2);
    end
end
